cmake_minimum_required(VERSION 3.24)
project(RecordBase VERSION 2.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(recordbase_core STATIC src/hierarchy.cpp src/dataset_engine.cpp)
target_include_directories(recordbase_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(recordbase_core PUBLIC cxx_std_23)
if(MSVC)
 target_compile_options(recordbase_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(recordbase_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(recordbase_hierarchy_demo src/hierarchy_demo.cpp)
target_link_libraries(recordbase_hierarchy_demo PRIVATE recordbase_core)
enable_testing()
add_executable(recordbase_tests tests/hierarchy_tests.cpp)
target_link_libraries(recordbase_tests PRIVATE recordbase_core)
add_test(NAME recordbase_hierarchy_tests COMMAND recordbase_tests)
